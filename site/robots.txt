# This is a mirror and downstream mirrors are welcome to take from it. The file
# tree is not restricted: robots.txt only asks clients that choose to read it,
# and wget -m happens to read it, so disallowing /distfiles/ would shut out
# anyone syncing over HTTP. Keeping individual files out of search results is
# handled by X-Robots-Tag, which download tools ignore.
#
# Syncing the whole tree over rsync costs both sides less:
#
#   rsync -a --delete rsync://distfiles.gentoozh.org/gentoo-zh/ /srv/mirror/
#
# Directory listings are one front end rendered per URL, so indexing them only
# yields many entries with identical content; /_ls/ is the interface it reads
# those listings from, not a page for people.

User-agent: *
Disallow: /_ls/
Disallow: /files/
Allow: /
